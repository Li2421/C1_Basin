"""Fixed goal-conditioned clearance-weighted 0.05 m Dijkstra fields."""
import numpy as np
from scipy.sparse import coo_matrix
from scipy.sparse.csgraph import dijkstra
from single_integrator.environment import point_segment_distance, segment_distance

SPACING=.05
CLEARANCE_TARGET=.15
WALL_WEIGHT=4.
TRACE_STEP=.025

class NavigationGrid:
 def __init__(self,geometry):
  self.geometry=geometry;c=geometry.config
  self.x=np.linspace(-c.half_length,c.half_length,round(2*c.half_length/SPACING)+1)
  self.y=np.linspace(-c.half_height,c.half_height,round(2*c.half_height/SPACING)+1)
  xx,yy=np.meshgrid(self.x,self.y);self.points=np.stack([xx,yy],-1);self.shape=xx.shape
  p=self.points.reshape(-1,2);self.valid=geometry.valid_points(p).reshape(self.shape)
  self.clearance=(point_segment_distance(p[:,None],geometry.walls[None,:,0],geometry.walls[None,:,1]).min(-1)-geometry.margin).reshape(self.shape)
  self.penalty=1+WALL_WEIGHT*np.maximum(0,CLEARANCE_TARGET-self.clearance)/CLEARANCE_TARGET
  ii=np.arange(len(p)).reshape(self.shape);rr=[];cc=[];vv=[]
  for dy,dx in [(0,1),(1,-1),(1,0),(1,1)]:
   ya=slice(0,self.shape[0]-dy);yb=slice(dy,self.shape[0]);xa=slice(max(0,-dx),min(self.shape[1],self.shape[1]-dx));xb=slice(max(0,dx),min(self.shape[1],self.shape[1]+dx))
   a=ii[ya,xa].ravel();b=ii[yb,xb].ravel();ok=self.valid.ravel()[a]&self.valid.ravel()[b];a=a[ok];b=b[ok]
   # Exact swept validity prevents diagonal corner cutting through occupied space.
   ok=segment_distance(p[a,None],p[b,None],geometry.walls[None,:,0],geometry.walls[None,:,1]).min(-1)>geometry.margin
   a=a[ok];b=b[ok];cost=SPACING*np.hypot(dx,dy)*.5*(self.penalty.ravel()[a]+self.penalty.ravel()[b])
   rr.extend([a,b]);cc.extend([b,a]);vv.extend([cost,cost])
  self.graph=coo_matrix((np.concatenate(vv),(np.concatenate(rr),np.concatenate(cc))),shape=(len(p),len(p))).tocsr()
 def field(self,goal):return NavigationField(self,goal)

class NavigationField:
 def __init__(self,grid,goal):
  self.grid=grid;self.geometry=grid.geometry;self.goal=np.asarray(goal,float)
  assert self.geometry.valid_points(self.goal[None])[0]
  p=grid.points.reshape(-1,2);dist=np.linalg.norm(p-self.goal,axis=1)
  nearby=np.flatnonzero(grid.valid.ravel()&(dist<=2*SPACING))
  visible=np.array([self.geometry.visible(self.goal,p[i]) for i in nearby]);nearby=nearby[visible]
  if not len(nearby):raise ValueError('No visible goal connection')
  penalty=1+WALL_WEIGHT*max(0,CLEARANCE_TARGET-(point_segment_distance(self.goal[None],self.geometry.walls[:,0],self.geometry.walls[:,1]).min()-self.geometry.margin))/CLEARANCE_TARGET
  initial=dist[nearby]*.5*(grid.penalty.ravel()[nearby]+penalty)
  # Multiple goal links through one virtual terminal node.
  n=len(p);base=grid.graph.tocoo();graph=coo_matrix((np.r_[base.data,initial,initial],(np.r_[base.row,nearby,np.full(len(nearby),n)],np.r_[base.col,np.full(len(nearby),n),nearby])),shape=(n+1,n+1)).tocsr()
  self.cost=dijkstra(graph,directed=False,indices=n)[:n].reshape(grid.shape)
 def _cell(self,p):
  xy=(p-np.array([self.grid.x[0],self.grid.y[0]]))/SPACING
  ij=np.floor(xy).astype(int);ij[:,0]=np.clip(ij[:,0],0,len(self.grid.x)-2);ij[:,1]=np.clip(ij[:,1],0,len(self.grid.y)-2)
  return ij,np.clip(xy-ij,0,1)
 def gradient(self,p):
  p=np.atleast_2d(p);ij,t=self._cell(p);x,y=ij.T;tx,ty=t.T
  a=self.cost[y,x];b=self.cost[y,x+1];c=self.cost[y+1,x];d=self.cost[y+1,x+1]
  good=np.isfinite(np.stack([a,b,c,d],-1)).all(-1)
  vals=np.where(np.isfinite(np.stack([a,b,c,d],-1)),np.stack([a,b,c,d],-1),0);a,b,c,d=vals.T
  g=np.stack([((1-ty)*(b-a)+ty*(d-c))/SPACING,((1-tx)*(c-a)+tx*(d-b))/SPACING],-1)
  return g,good
 def _fallback(self,p):
  # Steepest decrease among the eight grid neighbors; only visible links.
  ij=np.rint((p-np.array([self.grid.x[0],self.grid.y[0]]))/SPACING).astype(int)
  offsets=np.array([(dx,dy) for dy in [-1,0,1] for dx in [-1,0,1]])
  q=ij[:,None]+offsets;q[:,:,0]=np.clip(q[:,:,0],0,len(self.grid.x)-1);q[:,:,1]=np.clip(q[:,:,1],0,len(self.grid.y)-1)
  targets=self.grid.points[q[:,:,1],q[:,:,0]];cost=self.cost[q[:,:,1],q[:,:,0]]
  dist=np.linalg.norm(targets-p[:,None],axis=-1)
  safe=segment_distance(p[:,None,None,:],targets[:,:,None,:],self.geometry.walls[None,None,:,0],self.geometry.walls[None,None,:,1]).min(-1)>self.geometry.margin
  # Estimate cost at current position by its best visible neighbor connection.
  local=np.min(np.where(safe,cost+dist,np.inf),axis=1)
  slope=np.where(safe&(dist>1e-9),(local[:,None]-cost)/np.maximum(dist,1e-9),-np.inf)
  k=np.argmax(slope,axis=1);target=targets[np.arange(len(p)),k];v=target-p
  return v/np.maximum(np.linalg.norm(v,axis=1,keepdims=True),1e-30)
 def waypoint(self,positions,s):
  p=np.atleast_2d(np.asarray(positions,float)).copy();remaining=np.full(len(p),float(s))
  for _ in range(int(np.ceil(s/TRACE_STEP))+1):
   distance=np.linalg.norm(self.goal-p,axis=1);active=remaining>1e-9
   near=active&(distance<=np.minimum(remaining,2*SPACING))
   if near.any():
    ids=np.flatnonzero(near);ok=np.array([self.geometry.visible(p[i],self.goal) for i in ids]);ids=ids[ok];p[ids]=self.goal;remaining[ids]=0
   active=remaining>1e-9
   if not active.any():break
   ids=np.flatnonzero(active);a=p[ids];grad,valid=self.gradient(a);norm=np.linalg.norm(grad,axis=1,keepdims=True);direction=-grad/np.maximum(norm,1e-30)
   step=np.minimum(TRACE_STEP,remaining[ids]);b=a+step[:,None]*direction
   safe=self.geometry.valid_points(b)&(segment_distance(a[:,None],b[:,None],self.geometry.walls[None,:,0],self.geometry.walls[None,:,1]).min(-1)>self.geometry.margin)
   fallback=~valid|~safe|(norm[:,0]<1e-9)
   if fallback.any():direction[fallback]=self._fallback(a[fallback]);b=a+step[:,None]*direction
   p[ids]=b;remaining[ids]-=step
  return p

def expert_action(env,fields):
 p=env.positions;d=np.linalg.norm(env.goals-p,axis=1)
 target=np.concatenate([f.waypoint(p[i:i+1],.3) for i,f in enumerate(fields)])
 delta=target-p;u=delta/np.maximum(np.linalg.norm(delta,axis=1,keepdims=True),1e-30)*env.config.max_speed*np.minimum(1,d/.3)[:,None]
 u[d<=env.config.goal_tolerance]=0
 return u
